DETECTOR = (X, Y, S, \delExt, \delInt, \lambda, \ta)

\where

X = { (p,action) | p \in {"action_in"} \and action \in ACTIONS};
Y = { (p,signal) | p \in {"signal_out"} \and signal \in SIGNALS};
S = { (signal,sent) | signal \in SIGNALS \and sent \in {\true,\false}};

\delExt((signal,sent),e,(p,action)) = (signal',sent'),
	\if p \equalTo "action_in" \and action \in {"Typing","Waving"} \and signal \equalTo "Still" 	\then 	(signal',sent') \assign ("Moving",\false),
	\if p \equalTo "action_in" \and action \in {"Gone","Reading"} \and signal \equalTo "Moving" 	\then 	(signal',sent') \assign ("Still",\false),
	\if p \equalTo "action_in" 															\then 	(signal',sent') \assign (signal,\true);

\delInt((signal,sent)) = (signal,\true) 		\if sent \equalTo \false;

\lambda((signal,sent)) = ("signal_out",signal) 	\if sent \equalTo \false;

\ta((signal,sent)) = delta_int',
			\if \not sent	\then	delta_int' \assign 0,
			\if sent		\then	delta_int' \assign \infinity;
