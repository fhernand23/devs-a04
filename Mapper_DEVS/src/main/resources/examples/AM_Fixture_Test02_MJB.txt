FIXTURE(#delta_saving) = (X, Y, S, \delExt, \delInt, \lambda, \ta)

\where

X = {(p,signal) | p \in {"signal_in"} \and signal \in SIGNALS};
Y = {(p,level) | p \in {"level_out"}, level \in LEVELS};

S = {(level,sigma) | level \in LEVELS, sigma \in \R};

\delExt((level,sigma),e,(p,signal)) = 
	\if (p,signal) \equalTo ("signal_in","moving") \and level \equalTo "dark" 								\then	(level,0),
	\if (p,signal) \equalTo ("signal_in","still") \and level \equalTo "light" \and sigma \equalTo \infinity 		\then	(level,#delta_saving),
	\if (p,signal) \equalTo ("signal_in","still") \and level \equalTo "light" \and sigma \notEqualTo \infinity	\then	(level,sigma \substraction e),
 	\if p \equalTo "signal_in"																	\then	(level,\infinity);

\delInt((level,sigma)) = 
	\if (level \equalTo "dark")		\then	("light",\infinity),
	\if (level \equalTo "light")		\then	("dark",\infinity);

\lambda((level,sigma)) = 
	\if (level \equalTo "dark")		\then	("level_out","light"),	
	\if (level \equalTo "light")		\then	("level_out","dark");

\ta((level,sigma)) = sigma;
