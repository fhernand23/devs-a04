ENVIRONMENT(#delta_saving,#delta_typing,#delta_reading) = (X, Y, D, M, EIC, EOC, IC, \select)

\where

X = {(p,phase) | p \in {"phase_in"}, phase \in PHASES};
Y = Y_ACTION \union Y_LEVEL;
	Y_ACTION = {(p,action) | p \in {"action_out"}, action \in ACTIONS};
	Y_LEVEL = {(p,level) | p \in {"level_out"}, level \in LEVELS};

D = {detector,fixture,worker};

M = { (detector, DETECTOR), (fixture,FIXTURE(#delta_saving)), (worker, WORKER(#delta_typing,#delta_reading))};

EIC = {((ENVIRONMENT,"phase_in"),(worker,"phase_in"))};

EOC = {((fixture,"level_out"),(ENVIRONMENT,"level_out")),((worker,"action_out"),(ENVIRONMENT,"action_out"))};

IC = {((worker,"action_out"),(detector,"action_in")),((detector,"signal_out"),(fixture,"signal_in")),((fixture,"level_out"),(worker,"level_in"))};

\select(D_IMM) = ds',
	ds' \assign fixture 		\if (fixture \in D_IMM),
	ds' \assign detector		\if detector \in D_IMM \and fixture \notIn D_IMM,
	ds' \assign worker	 		\if worker \in D_IMM \and detector \notIn D_IMM \and fixture \notIn D_IMM;
