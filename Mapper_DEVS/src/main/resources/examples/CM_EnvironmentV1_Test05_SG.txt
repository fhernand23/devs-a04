Environment = (X, Y, D, M, EIC, EOC, IC, \select)

\where

X = {(phasein,phase) | phase \in PHASES};
PHASES = {"Present", "Absent"};
Y = YACTION \union YLEVEL;
YACTION = {(actionout,action) | action \in ACTIONS};
ACTIONS = {"Gone", "Typing", "Waving", "Reading"};
YLEVEL = {(levelout,level) | level \in LEVELS};
LEVELS = {"Dark", "Light"};
D = {"Detector", "Fixture", "Worker"};
M = {("Detector", Detector), ("Fixture", Fixture), ("Worker", Worker)};


EIC = {(("OUT",phasein),("Worker", phasein))};
EOC = {(("Fixture", levelout),("OUT",levelout)),(("Worker",actionout),("OUT",actionout))};
IC = {(("Worker", actionout),("Detector",actionin)),(("Detector",signalout),("Fixture",signalin)),(("Fixture",levelout),("Worker",levelin))};
