ENVIRONMENT(#delta_saving,#delta_typing,#delta_reading) = (X, Y, D, M, EIC, EOC, IC, \select)

\where

X = {(p,phase) | p \in {"phase_in"}, phase \in PHASES};
Y = Y_ACTION \union Y_LEVEL;
	Y_ACTION = {(p,action) | p \in {"action_out"}, action \in ACTIONS};
	Y_LEVEL = {(p,level) | p \in {"level_out"}, level \in LEVELS};

D = {lighting,worker};

M = { (lighting, LIGHTING(#delta_saving)), (worker, WORKER(#delta_typing,#delta_reading))};

EIC = {((ENVIRONMENT,"phase_in"),(worker,"phase_in"))};

EOC = {((lighting,"level_out"),(ENVIRONMENT,"level_out")),((worker,"action_out"),(ENVIRONMENT,"action_out"))};

IC = {((worker,"action_out"),(lighting,"action_in")),((lighting,"level_out"),(worker,"level_in"))};

\select(D_IMM) =
	lighting		\if (lighting \in D_IMM),
	worker		\if worker \in D_IMM \and lighting \notIn D_IMM;
