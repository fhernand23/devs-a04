FIXTURE(#delta_saving) = (X, Y, S, \delExt, \delInt, \lambda, \ta)

\where

X = {(p,signal) | p \in {"signal_in"} \and signal \in SIGNALS};
Y = {(p,level) | p \in {"level_out"}, level \in LEVELS};

S = {(level,sigma) | level \in LEVELS, sigma \in \R};

\delExt((level,sigma),e,(p,signal)) = (level,sigma'),
	\if (p,signal) \equalTo ("signal_in","moving") \and level \equalTo "dark" 								\then	sigma' \assign 0,
	\if (p,signal) \equalTo ("signal_in","still") \and level \equalTo "light" \and sigma \equalTo \infinity 		\then	sigma' \assign #delta_saving,
	\if (p,signal) \equalTo ("signal_in","still") \and level \equalTo "light" \and sigma \notEqualTo \infinity	\then	sigma' \assign sigma \substraction e,
 	\if p \equalTo "signal_in"																		\then	sigma' \assign \infinity;

\delInt((level,sigma)) = (level',\infinity),
	\if (level \equalTo "dark")		\then	level' \assign "light",
	\if (level \equalTo "light")		\then	level' \assign "dark";

\lambda((level,sigma)) = ("level_out",level'),
	\if (level \equalTo "dark")		\then	level' \assign "light",	
	\if (level \equalTo "light")		\then	level' \assign "dark";

\ta((level,sigma)) = sigma;
