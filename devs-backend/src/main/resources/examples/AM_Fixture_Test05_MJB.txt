FIXTURE(#delta_saving) = (X, Y, S, \delExt, \delInt, \lambda, \ta)

\where

X = {(p,signal) | p \in {"signal_in"} \and signal \in SIGNALS};
Y = {(p,level) | p \in {"level_out"}, level \in LEVELS};

S = {(level,sigma) | level \in LEVELS, sigma \in \R};

\delExt((level,sigma),e,(p,signal)) = (level,sigma'),
	sigma' \assign 0 				\if p \equalTo "signal_in" \and signal \equalTo "moving" \and level \equalTo "dark",
	sigma' \assign #delta_saving		\if p \equalTo "signal_in" \and signal \equalTo "still" \and level \equalTo "light" \and sigma \equalTo \infinity,
	sigma' \assign (sigma \substraction e)	\if p \equalTo "signal_in" \and signal \equalTo "still"\and level \equalTo "light" \and sigma \notEqualTo \infinity,
 	sigma' \assign \infinity 			\if p \equalTo "signal_in";

\delInt((level,sigma)) = (level', \infinity),
	level' \assign "light"			\if (level \equalTo "dark"),
	level' \assign "dark"			\if (level \equalTo "light");

\lambda((level,sigma)) = ("level_out",level'),
	level' \assign "light"		\if (level \equalTo "dark"),
	level' \assign "dark"		\if (level \equalTo "light");

\ta((level,sigma)) = sigma;
