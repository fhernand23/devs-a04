WORKER(#delta_typing,#delta_reading)= (X, Y, S, \delExt, \delInt, \lambda, \ta)

\where

X = X_PHASE \union X_LEVEL;
	X_PHASE = {(p,phase) | p \in {"phase_in"} \and phase \in PHASES };
	X_LEVEL = {(p,level) | p \in {"level_in"}, level \in LELEVLS };

Y = {(p,action) | p \in {"action_out"}, action \in ACTIONS};

S = { (action,delta_t_r,sent) | action \in ACTIONS, delta_t_r \in \R, sent \in \B};

\delExt((action,delta_t_r,sent),e,(p,v)) = 
	("Typing",#delta_typing,\false)					\if p \equalTo "phase_in" \and v \equalTo "Present" \and action \equalTo "Gone",
	("Gone",\infinity,\false)						\if p \equalTo "phase_in" \and v \equalTo "Absent" \and action \notEqualTo "Gone",
	("Waiving", delta_t_r \substraction e,\false)		\if p \equalTo "level_in" \and v \equalTo "Dark" \and action \equalTo "Reading",
	("Reading", delta_t_r \substraction e,\false)		\if p \equalTo "level_in" \and v \equalTo "Light" \and action \equalTo "Waiving",
	(action,delta_t_r \substraction e,sent)			\inOtherCase;

\delInt((action,delta_t_r,sent)) = 
	(action, delta_t_r,\true) 		\if \not sent,
	("Reading",#delta_reading,\true) 	\if action \equalTo "Typing" \and sent,
	("Typing",#delta_typing,\true) 	\if action \in {"Reading","Waving"} \and sent;

\lambda((action,delta_t_r,sent)) = 
	("action_out",action)		\if \not sent,
	("action_out","Reading")	\if action \equalTo "Typing" \and sent,
	("action_out","Typing")		\if action \in {"Reading","Waving"} \and sent;

\ta((action,delta_t_r,sent)) = 
	0			\if \not sent,
	delta_t_r 	\if sent;
