WORKER(#delta_typing,#delta_reading) = (X, Y, S, \delExt, \delInt, \lambda, \ta)

\where

X = X_PHASE \union X_LEVEL;
	X_PHASE = {(p,phase) | p \in {"phase_in"} \and phase \in PHASES };
	X_LEVEL = {(p,level) | p \in {"level_in"}, level \in LELEVLS };

Y = {(p,action) | p \in {"action_out"}, action \in ACTIONS};

S = { (action,delta_t_r,sent) | action \in ACTIONS, delta_t_r \in \R, sent \in \B};

\delExt((action,delta_t_r,sent),e,(p,v)) = 
	\if (p,v) \equalTo ("phase_in","Present") \and action \equalTo "Gone"		\then	("Typing",#delta_typing,\false),				
	\if (p,v) \equalTo ("phase_in","Absent") \and action \notEqualTo "Gone"			\then	("Gone",\infinity,\false),
	\if (p,v) \equalTo ("level_in","Dark") \and action \equalTo "Reading"			\then	("Waiving", delta_t_r \substraction e,\false),
	\if (p,v) \equalTo ("level_in","Light") \and action \equalTo "Waiving"			\then	("Reading", delta_t_r \substraction e,\false),
	\inOtherCase																			(action,delta_t_r \substraction e,sent);

\delInt((action,delta_t_r,sent)) = 
	\if \not sent							\then		(action, delta_t_r,\true),
	\if action \equalTo "Typing" \and sent		\then		("Reading",#delta_reading,\true),
	\if action \in {"Reading","Waving"} \and sent	\then	("Typing",#delta_typing,\true);

\lambda((action,delta_t_r,sent)) = 
	\if \not sent								\then	("action_out",action),
	\if action \equalTo "Typing" \and sent			\then	("action_out","Reading"),			
	\if action \in {"Reading","Waving"} \and sent	\then	("action_out","Typing");

\ta((action,delta_t_r,sent)) = 
	\if \not sent		\then	0,
	\if sent			\then	delta_t_r;
